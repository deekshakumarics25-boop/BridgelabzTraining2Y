SECTION 1: PSEUDOCODE PROBLEMS

PC-1: Prime Number Checker

 PrimeNumberChecker
    INPUT N

    Set isPrime = TRUE

    FOR i = 2 TO sqrt(N)
        IF N MOD i = 0 THEN
            Set isPrime = FALSE
            BREAK
        END IF
    END FOR

    IF isPrime = TRUE THEN
        OUTPUT "Prime"
    ELSE
        OUTPUT "Not Prime"
    END IF
End 

PC-2: Number Digit Reversal

 ReverseNumber
    INPUT N

    Set reversed = 0

    WHILE N > 0
        digit = N MOD 10
        reversed = reversed * 10 + digit
        N = N / 10     
    END WHILE

    OUTPUT reversed
End
